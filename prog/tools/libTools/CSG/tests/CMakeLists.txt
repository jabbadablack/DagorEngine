cmake_minimum_required(VERSION 3.20)
project(dagor_csg_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
if(MSVC)
  add_compile_options(/MP)
endif()
get_filename_component(PROG "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(CARVE "${PROG}/3rdPartyLibs/math/carve")
file(GLOB CARVE_SOURCES "${CARVE}/lib/*.cpp")
list(FILTER CARVE_SOURCES EXCLUDE REGEX "intersect_classify_simple.cpp$")
add_library(csg_carve STATIC ${CARVE_SOURCES})
target_include_directories(csg_carve PUBLIC "${CARVE}/include")
target_compile_definitions(csg_carve PUBLIC CARVE_USE_EXACT_PREDICATES CARVE_NODEBUG NDEBUG)
if(WIN32)
  target_compile_definitions(csg_carve PUBLIC WIN32)
endif()
add_executable(csg_tests main.cpp ../brush.cpp ../brushCarve.cpp)
target_include_directories(csg_tests PRIVATE "${PROG}/tools/sharedInclude")
target_link_libraries(csg_tests PRIVATE csg_carve)
enable_testing()
add_test(NAME csg_regressions COMMAND csg_tests)
add_executable(csg_benchmark benchmark.cpp ../brush.cpp ../brushCarve.cpp)
target_include_directories(csg_benchmark PRIVATE "${PROG}/tools/sharedInclude")
target_link_libraries(csg_benchmark PRIVATE csg_carve)
